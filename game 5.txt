<script>
    const canvas = document.getElementById('game-canvas');
    const ctx = canvas.getContext('2d');

    let quizBank = [
        { q: "成人每日建議飲水量是體重(kg)乘以多少？", options: ["10ml", "20ml", "30-35ml", "100ml"], correct: 2 },
        { q: "下列哪種運動最能增進心肺功能？", options: ["看電視", "慢跑", "睡覺", "滑手機"], correct: 1 },

        { q: "正確刷牙法「貝氏刷牙法」，刷毛與牙齒夾角？", options: ["15度", "45度", "90度", "180度"], correct: 1 },

        { q: "預防骨質疏鬆，應該多補充哪種營養素？", options: ["蛋白質", "鈣質", "脂肪", "糖分"], correct: 1 },
        { q: "二手菸對身體有害嗎？", options: ["沒影響", "比一手菸毒", "只有味道臭", "會變聰明"], correct: 1 },

        { q: "所謂「天天五蔬果」是指？", options: ["5顆蘋果", "3份蔬菜2份水果", "5杯果汁", "只吃水果"], correct: 1 },
        { q: "哪種行為最容易傳播流感？", options: ["打噴嚏不遮口鼻", "常洗手", "戴口罩", "均衡飲食"], correct: 0 },
        { q: "BMI指數是用來測量什麼？", options: ["智力", "體重是否標準", "視力", "聽力"], correct: 1 },
        { q: "哪種飲料最健康？", options: ["珍珠奶茶", "純水", "能量飲", "碳酸飲料"], correct: 1 },
        { q: "預防腸病毒最有效的方法？", options: ["不睡覺", "勤洗手", "多看書", "穿厚衣服"], correct: 1 },
        { q: "中暑時應該優先做什麼？", options: ["跑步", "移至陰涼處", "穿大衣", "喝熱咖啡"], correct: 1 },
        { q: "哪種維生素跟眼睛健康最相關？", options: ["維生素A", "維生素B12", "維生素E", "維生素K"], correct: 0 },
        { q: "如廁後哪項行為最重要？", options: ["洗手", "梳頭", "照鏡子", "唱歌"], correct: 0 },
        { q: "均衡飲食中，哪類食物應吃最多？", options: ["油脂類", "全穀雜糧類", "糖果類", "炸雞類"], correct: 1 }
    ];

    let currentLevel = 0;
    let score = 0;
    let hp = 3;
    let doorOpen = false;
    let gameStarted = false;
    let isWaiting = false;
    let animationId = null;
    let platforms = [];
    let coins = [];
    let wrongAnswers = [];
    let stars = [];

    const player = {
        x: 50, y: 50, vx: 0, vy: 0, size: 24,
        onGround: false, speed: 3.8, jumpPower: -10, gravity: 0.45
    };

    const keys = {};

    // 初始化背景星星
    for(let i=0; i<80; i++) {
        stars.push({ x: Math.random()*800, y: Math.random()*520, size: Math.random()*2, opacity: Math.random() });
    }

    function startGame() {
        if (gameStarted) return;
        document.getElementById('start-screen').style.display = 'none';
        quizBank.sort(() => Math.random() - 0.5);
        gameStarted = true;
        initLevel();
        if (animationId) cancelAnimationFrame(animationId);
        gameLoop();
    }

    function generateMap() {
        // 平台設計：包含斷開與移動平台
        platforms = [
            { x: 0, y: 130, w: 200, h: 12, moveX: 0, moveY: 0, range: 0, speed: 0 },
            { x: 280, y: 130, w: 200, h: 12, moveX: 0, moveY: 0, range: 0, speed: 0 },
            { x: 550, y: 130, w: 250, h: 12, moveX: 0, moveY: 0, range: 0, speed: 0 },
            
            // 中層移動平台 (左右移動)
            { x: 100, y: 280, w: 150, h: 12, moveX: 1, moveY: 0, range: 150, speed: 0.02, offset: 0 },
            { x: 500, y: 280, w: 150, h: 12, moveX: 1, moveY: 0, range: 100, speed: 0.03, offset: Math.PI },
            
            // 底層斷開平台
            { x: 0, y: 440, w: 300, h: 15, moveX: 0, moveY: 0, range: 0, speed: 0 },
            { x: 450, y: 440, w: 350, h: 15, moveX: 0, moveY: 0, range: 0, speed: 0 }
        ];

        // 紀錄初始位置供移動參考
        platforms.forEach(p => { p.startX = p.x; p.startY = p.y; });

        coins = []; 
        const data = quizBank[currentLevel];
        
        // 隨機在平台上放置硬幣
        let availablePlatforms = [...platforms];
        availablePlatforms.sort(() => Math.random() - 0.5);

        data.options.forEach((opt, i) => {
            const p = availablePlatforms[i % availablePlatforms.length];
            const cx = p.x + Math.random() * (p.w - 30);
            const cy = p.y - 45;
            coins.push({ x: cx, y: cy, text: opt, isCorrect: i === data.correct, collected: false, parentPlatform: p });
        });

        player.x = 40; player.y = 80;
        player.vx = 0; player.vy = 0;
    }

    function initLevel() {
        doorOpen = false;
        isWaiting = true;
        if (currentLevel > 0 && currentLevel % 5 === 0 && hp < 3) hp++;
        generateMap();
        updateUI();
    }

    function updateUI() {
        document.getElementById('q-level').innerText = `MYSTERY STEP ${currentLevel + 1} / 20`;
        document.getElementById('q-text').innerText = quizBank[currentLevel].q;
        document.getElementById('question-box').style.borderColor = "#4facfe";
        document.getElementById('status-hp').innerText = "HP: " + "❤️".repeat(hp);
        document.getElementById('status-score').innerText = "SCORE: " + score;
    }

    function showReviewScreen(title) {
        gameStarted = false;
        document.getElementById('review-screen').style.display = 'flex';
        document.getElementById('review-title').innerText = title;
        document.getElementById('final-score').innerText = "最終得分: " + score;
        const listDiv = document.getElementById('review-list');
        if (wrongAnswers.length === 0) {
            listDiv.innerHTML = "<p>你已洞悉所有真相！完美通關。</p>";
        } else {
            listDiv.innerHTML = wrongAnswers.map(item => `
                <div class="review-item">
                    <strong>問：${item.q}</strong><br>
                    <span style="color: #00f2fe;">正確答案：${item.a}</span>
                </div>
            `).join('');
        }
    }

    function loseHp() {
        const currentQ = quizBank[currentLevel];
        if (!wrongAnswers.find(item => item.q === currentQ.q)) {
            wrongAnswers.push({ q: currentQ.q, a: currentQ.options[currentQ.correct] });
        }
        hp--;
        if (hp <= 0) {
            showReviewScreen("挑戰結束");
        } else {
            updateUI();
            player.x = 40; player.y = 80; player.vx = 0; player.vy = 0;
            isWaiting = true;
        }
    }

    window.addEventListener('keydown', e => { 
        if(gameStarted) {
            if(isWaiting && e.code === 'Space') isWaiting = false;
            keys[e.code] = true; 
        }
    });
    window.addEventListener('keyup', e => { if(gameStarted) keys[e.code] = false; });

    canvas.addEventListener('touchstart', (e) => {
        if(gameStarted && isWaiting) { isWaiting = false; e.preventDefault(); }
    });

    const btnMap = { 'btn-left': 'ArrowLeft', 'btn-right': 'ArrowRight', 'btn-jump': 'ArrowUp' };
    Object.keys(btnMap).forEach(id => {
        const el = document.getElementById(id);
        el.addEventListener('touchstart', (e) => { 
            e.preventDefault(); 
            if(gameStarted) {
                if(isWaiting) isWaiting = false;
                keys[btnMap[id]] = true; 
            }
        });
        el.addEventListener('touchend', (e) => { e.preventDefault(); if(gameStarted) keys[btnMap[id]] = false; });
    });

    function update() {
        if (!gameStarted || isWaiting) return;

        // 更新平台位置
        platforms.forEach(p => {
            if(p.moveX) {
                p.x = p.startX + Math.sin(Date.now() * p.speed + (p.offset || 0)) * p.range;
            }
        });

        // 玩家左右移動
        if (keys['ArrowLeft']) player.vx = -player.speed;
        else if (keys['ArrowRight']) player.vx = player.speed;
        else player.vx *= 0.8;

        // 玩家跳躍
        if (keys['ArrowUp'] && player.onGround) {
            player.vy = player.jumpPower;
            player.onGround = false;
        }

        player.vy += player.gravity;
        player.x += player.vx;
        player.y += player.vy;
        player.onGround = false;

        // 平台碰撞
        platforms.forEach(p => {
            if (player.x + player.size > p.x && player.x < p.x + p.w &&
                player.y + player.size > p.y && player.y + player.size < p.y + p.h + 12 && player.vy >= 0) {
                player.y = p.y - player.size;
                player.vy = 0;
                player.onGround = true;
                // 玩家隨移動平台移動
                if(p.moveX) player.x += Math.cos(Date.now() * p.speed + (p.offset || 0)) * p.range * p.speed * 2; 
                // 簡化處理：直接加上位移
                let dx = Math.cos(Date.now() * p.speed + (p.offset || 0)) * p.range * p.speed * 1.5;
                if(p.moveX) player.x += dx;
            }
        });

        // 硬幣碰撞與位置更新
        coins.forEach(c => {
            if (!c.collected) {
                // 如果硬幣附著在移動平台上
                if(c.parentPlatform && c.parentPlatform.moveX) {
                    c.x = c.parentPlatform.x + (c.pOffset || 20);
                } else if(!c.pOffset) {
                    c.pOffset = c.x - (c.parentPlatform ? c.parentPlatform.x : 0);
                }

                if (Math.abs(player.x - c.x) < 35 && Math.abs(player.y - c.y) < 45) {
                    c.collected = true;
                    if (c.isCorrect) {
                        doorOpen = true; score += 100;
                        document.getElementById('q-text').innerText = "正解！幻境之門已開啟！";
                        document.getElementById('question-box').style.backgroundColor = "rgba(46, 204, 113, 0.3)";
                    } else { 
                        score = Math.max(0, score - 20); loseHp(); 
                    }
                }
            }
        });

        if (player.y > 520) loseHp();
        if (player.x < 0) player.x = 0;
        if (player.x > 800 - player.size) player.x = 800 - player.size;

        if (doorOpen && player.x > 710 && player.y > 360) {
            currentLevel++;
            if (currentLevel < quizBank.length) initLevel();
            else showReviewScreen("傳說達成！你已征服幻境。");
        }
    }

    function draw() {
        // 繪製神祕風格背景
        let grad = ctx.createLinearGradient(0, 0, 0, 520);
        grad.addColorStop(0, "#0f0c29");
        grad.addColorStop(0.5, "#302b63");
        grad.addColorStop(1, "#24243e");
        ctx.fillStyle = grad;
        ctx.fillRect(0, 0, 800, 520);

        // 畫星星
        ctx.fillStyle = "white";
        stars.forEach(s => {
            ctx.globalAlpha = Math.abs(Math.sin(Date.now()*0.001 + s.opacity)*s.opacity);
            ctx.fillRect(s.x, s.y, s.size, s.size);
        });
        ctx.globalAlpha = 1.0;
        
        // 繪製平台
        platforms.forEach(p => {
            let pGrad = ctx.createLinearGradient(p.x, p.y, p.x, p.y + p.h);
            pGrad.addColorStop(0, "#4facfe");
            pGrad.addColorStop(1, "#00f2fe");
            ctx.fillStyle = pGrad;
            ctx.shadowBlur = 10;
            ctx.shadowColor = "#4facfe";
            ctx.fillRect(p.x, p.y, p.w, p.h);
            ctx.shadowBlur = 0;
        });
        
        // 繪製硬幣 (發光金幣)
        coins.forEach(c => {
            if (!c.collected) {
                ctx.shadowBlur = 15;
                ctx.shadowColor = "#f1c40f";
                ctx.fillStyle = "#f1c40f";
                ctx.beginPath(); ctx.arc(c.x+15, c.y+15, 12, 0, Math.PI*2); ctx.fill();
                ctx.shadowBlur = 0;
                ctx.fillStyle = "white"; ctx.font = "bold 14px Arial"; ctx.textAlign = "center";
                ctx.fillText(c.text, c.x + 15, c.y - 15);
            }
        });

        // 繪製傳送門 (霓虹效果)
        if (doorOpen) {
            ctx.shadowBlur = 20;
            ctx.shadowColor = "#00f2fe";
            ctx.fillStyle = (Date.now() % 400 > 200) ? "#00f2fe" : "#4facfe";
            ctx.fillRect(720, 370, 50, 70);
            ctx.shadowBlur = 0;
        } else {
            ctx.fillStyle = "rgba(255,255,255,0.1)";
            ctx.fillRect(720, 370, 50, 70);
            ctx.strokeStyle = "#4facfe";
            ctx.strokeRect(720, 370, 50, 70);
        }

        // 繪製主角 (能量方塊)
        ctx.shadowBlur = 15;
        ctx.shadowColor = "#4ea5ff";
        ctx.fillStyle = "#fff";
        ctx.fillRect(player.x, player.y, player.size, player.size);
        ctx.fillStyle = "#4ea5ff";
        ctx.fillRect(player.x+4, player.y+4, player.size-8, player.size-8);
        ctx.shadowBlur = 0;

        if (isWaiting) {
            ctx.fillStyle = "rgba(0, 0, 0, 0.7)";
            ctx.fillRect(0, 0, canvas.width, canvas.height);
            ctx.fillStyle = "#00f2fe";
            ctx.font = "bold 32px Arial";
            ctx.textAlign = "center";
            ctx.fillText("🌌 準備踏入下一層幻境 🌌", canvas.width / 2, canvas.height / 2 - 20);
            ctx.font = "20px Arial";
            ctx.fillStyle = "white";
            ctx.fillText("按下空白鍵或點擊螢幕開始", canvas.width / 2, canvas.height / 2 + 30);
        }
    }

    function gameLoop() {
        if (!gameStarted) return;
        update();
        draw();
        animationId = requestAnimationFrame(gameLoop);
    }
</script>
